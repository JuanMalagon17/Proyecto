Factory Method — DaoFactory e InMemoryDaoFactory

Problema: El sistema requiere aislar la persistencia física de datos (catálogos de productos) para facilitar las pruebas automatizadas veloces sin alterar los archivos JSON de producción.

from abc import ABC, abstractmethod
from src.dao.interface_dao import IDao
from src.dao.producto_dao import ProductoDao
from src.models.producto import Producto

class DaoFactory(ABC):
    """Fábrica abstracta de DAOs — Factory Method."""
    @abstractmethod
    def crear_producto_dao(self) -> IDao: ...
    @abstractmethod
    def crear_factura_dao(self) -> IDao: ...
    @abstractmethod
    def crear_cliente_dao(self) -> IDao: ...

class InMemoryDaoFactory(DaoFactory):
    """Implementación de pruebas unitarias con almacenamiento en diccionario."""
    def __init__(self) -> None:
        self._db = _InMemoryStore()
    def crear_producto_dao(self) -> IDao: return self._db
    def crear_factura_dao(self) -> IDao: return self._db
    def crear_cliente_dao(self) -> IDao: return self._db

class _InMemoryStore(IDao):
    def __init__(self):
        self._store = {}
    def guardar(self, e):
        uid = getattr(e, "id_producto", getattr(e, "id_factura", getattr(e, "id_cliente", None)))
        self._store[uid] = e
        return e
    def buscar_por_id(self, id_): return self._store.get(id_)
    def actualizar(self, e): return self.guardar(e)

Singleton & Decorator — ConfiguracionApp y ValidarLimiteCreditoDecorator

Problema: Asegurar que variables de control de riesgo comercial se almacenen en una única instancia thread-safe, interceptando el guardado de facturas para denegar transacciones que excedan el crédito asignado.

import threading
from src.dao.interface_dao import IDao
from src.models.factura import Factura

class ConfiguracionApp:
    """Singleton thread-safe para variables globales."""
    _instancia = None
    _lock = threading.Lock()

    def __new__(cls):
        if cls._instancia is None:
            with cls._lock:
                if cls._instancia is None:
                    cls._instancia = super().__new__(cls)
                    cls._instancia._config = {"limite_defecto": 5000000.0}
        return cls._instancia
    
    def obtener(self, clave: str): return self._config.get(clave)

class ValidarLimiteCreditoDecorator(IDao):
    """Decorator estructural para inyectar reglas de negocio sobre el DAO real."""
    def __init__(self, dao_real: IDao, limite_credito: float) -> None:
        self._dao = dao_real
        self._limite = limite_credito

    def guardar(self, factura: Factura) -> Factura:
        if factura.total > self._limite:
            raise ValueError(f"Operación bloqueada: Supera el límite de crédito de ${self._limite:,.2f}")
        return self._dao.guardar(factura)

Builder — FacturaBuilder

Problema: La instanciación nativa de una factura involucra cálculos de fechas de emisión, ensamble ordenado de líneas de ítems y validaciones de stock en tiempo real que sobrecargan el constructor base.

from src.models.factura import Factura, LineaFactura
from datetime import date

class FacturaBuilder:
    """Builder fluido para orquestar la construcción limpia de comprobantes de cartera."""
    def __init__(self) -> None: self._reset()
    def _reset(self) -> None:
        self._id_factura = ""; self._id_cliente = ""; self._nombre_cliente = ""
        self._fecha_emision = date.today().isoformat(); self._fecha_vencimiento = ""
        self._lineas = []

    def set_identificador(self, id_factura: str): self._id_factura = id_factura; return self
    def set_cliente(self, id_cl: str, nom: str): self._id_cliente = id_cl; self._nombre_cliente = nom; return self
    def set_vencimiento(self, fecha: str): self._fecha_vencimiento = fecha; return self

    def agregar_linea(self, producto, cantidad: int):
        if not producto.hay_stock_suficiente(cantidad):
            raise ValueError(f"Stock insuficiente para '{producto.nombre}'.")
        linea = LineaFactura(
            id_producto=producto.id_producto,
            nombre_producto=producto.nombre,
            cantidad=cantidad,
            precio_unitario=producto.precio_unitario,
            costo_unitario=producto.costo_unitario 
        )
        self._lineas.append(linea)
        return self

    def build(self) -> Factura:
        if not self._id_factura or not self._fecha_vencimiento:
            raise ValueError("Datos obligatorios faltantes para construir la Factura.")
        f = Factura(self._id_factura, self._id_cliente, self._nombre_cliente, self._fecha_emision, self._fecha_vencimiento)
        for l in self._lineas: f.agregar_linea(l)
        res = f; self._reset()
        return res

Facade & Command — GestionCarteraFacade y ComandoPagarFactura

Demostración de Trabajo Grupal: Integración cruzada. La clase GestionCarteraFacade toma los componentes de los tres desarrolladores y los expone mediante un punto de acceso unificado.

from abc import ABC, abstractmethod

class ComandoFactura(ABC):
    @abstractmethod
    def ejecutar(self) -> None: ...

class ComandoPagarFactura(ComandoFactura):
    def __init__(self, dao, id_factura: str) -> None:
        self._dao = dao; self._id = id_factura
    def ejecutar(self) -> None:
        factura = self._dao.buscar_por_id(self._id)
        if factura:
            factura.pagar()
            self._dao.actualizar(factura)

class GestionCarteraFacade:
    """Orquestador de Fachada única del sistema."""
    def __init__(self, factory_dao: DaoFactory) -> None:
        self._factura_dao = factory_dao.crear_factura_dao()
        self._cliente_dao = factory_dao.crear_cliente_dao()
        self._producto_dao = factory_dao.crear_producto_dao()

    def procesar_recaudo_caja(self, id_factura: str) -> bool:
        cmd = ComandoPagarFactura(self._factura_dao, id_factura)
        cmd.ejecutar()
        return True
